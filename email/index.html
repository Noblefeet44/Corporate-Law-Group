Run this clean script in your Supabase SQL Editor to create the emails and email_senders tables, enable Realtime, and set up Row Level Security. 

-- 1. Create the emails table
CREATE TABLE IF NOT EXISTS public.emails (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    message_id TEXT,
    from_address TEXT NOT NULL,
    from_name TEXT,
    to_address TEXT NOT NULL,
    to_name TEXT,
    cc TEXT,
    bcc TEXT,
    reply_to TEXT,
    subject TEXT DEFAULT '(no subject)',
    snippet TEXT,
    body_html TEXT,
    body_text TEXT,
    folder TEXT NOT NULL DEFAULT 'inbox',
    is_read BOOLEAN NOT NULL DEFAULT false,
    is_starred BOOLEAN NOT NULL DEFAULT false,
    has_attachment BOOLEAN NOT NULL DEFAULT false,
    attachments JSONB DEFAULT '[]'::jsonb,
    labels JSONB DEFAULT '["Inbox"]'::jsonb,
    raw_headers JSONB DEFAULT '{}'::jsonb,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_emails_folder ON public.emails(folder);
CREATE INDEX IF NOT EXISTS idx_emails_created_at ON public.emails(created_at DESC);
CREATE INDEX IF NOT EXISTS idx_emails_is_read ON public.emails(is_read);
CREATE INDEX IF NOT EXISTS idx_emails_is_starred ON public.emails(is_starred);
CREATE INDEX IF NOT EXISTS idx_emails_to_address ON public.emails(to_address);
CREATE INDEX IF NOT EXISTS idx_emails_from_address ON public.emails(from_address);

-- 2. Create the email_senders table (Sender profiles & aliases)
CREATE TABLE IF NOT EXISTS public.email_senders (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    display_name TEXT NOT NULL,
    email_address TEXT NOT NULL UNIQUE,
    is_default BOOLEAN NOT NULL DEFAULT false,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

INSERT INTO public.email_senders (display_name, email_address, is_default)
VALUES 
    ('Corporate Law Group Inquiries', 'inquiries@mail.corporatelawgroup.org', true),
    ('Corporate Law Group Support', 'support@mail.corporatelawgroup.org', false)
ON CONFLICT (email_address) DO NOTHING;

-- 3. Enable RLS
ALTER TABLE public.emails ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.email_senders ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow anon read all emails" ON public.emails FOR SELECT TO anon, authenticated, service_role USING (true);
CREATE POLICY "Allow anon insert emails" ON public.emails FOR INSERT TO anon, authenticated, service_role WITH CHECK (true);
CREATE POLICY "Allow anon update emails" ON public.emails FOR UPDATE TO anon, authenticated, service_role USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon delete emails" ON public.emails FOR DELETE TO anon, authenticated, service_role USING (true);

CREATE POLICY "Allow anon read senders" ON public.email_senders FOR SELECT TO anon, authenticated, service_role USING (true);
CREATE POLICY "Allow anon insert senders" ON public.email_senders FOR INSERT TO anon, authenticated, service_role WITH CHECK (true);
CREATE POLICY "Allow anon update senders" ON public.email_senders FOR UPDATE TO anon, authenticated, service_role USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon delete senders" ON public.email_senders FOR DELETE TO anon, authenticated, service_role USING (true);

-- 4. Enable Realtime
DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_publication_tables WHERE pubname = 'supabase_realtime' AND tablename = 'emails') THEN
    ALTER PUBLICATION supabase_realtime ADD TABLE public.emails;
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_publication_tables WHERE pubname = 'supabase_realtime' AND tablename = 'email_senders') THEN
    ALTER PUBLICATION supabase_realtime ADD TABLE public.email_senders;
  END IF;
END $$;

How to receive emails from @mail.corporatelawgroup.org via Resend into Supabase:

  1. Deploy the Supabase Edge Function provided at email/resend_inbound_function.ts using Supabase CLI:
    supabase functions deploy resend-inbound --no-verify-jwt
  2. In your Resend Dashboard → Webhooks, add an endpoint:
    https://xsnfafxcbdkyinytjhci.supabase.co/functions/v1/resend-inbound
  3. Select the email.received event.
  4. Every incoming email to inquiries@mail.corporatelawgroup.org, support@mail.corporatelawgroup.org, or custom aliases (e.g. pecker@mail.corporatelawgroup.org) will immediately insert into Supabase and pop up in your inbox in real-time!
